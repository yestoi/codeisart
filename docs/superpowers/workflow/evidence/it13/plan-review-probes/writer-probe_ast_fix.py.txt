def review_test(text):
    tree = ast.parse(text)
    attrs = [n for n in ast.walk(tree) if isinstance(n, ast.Attribute)]
    for name in ("push", "set_brightness", "close"):
        used = [n for n in attrs if n.attr == name]
        if not (used and all(ast.unparse(n.value) == "wall" for n in used)):
            return f"fails on {name}: {[ast.unparse(n) for n in used if ast.unparse(n.value) != 'wall']}"
    bad = [ast.unparse(n) for n in attrs if n.attr in ("_send", "display")]
    if bad:
        return f"fails on _send/display: {bad}"
    wraps = [n for n in ast.walk(tree) if isinstance(n, ast.Call) and isinstance(n.func, ast.Name)
             and n.func.id == "GovernedDisplay"]
    if not (len(wraps) == 1 and ast.unparse(wraps[0].args[0]) == "display"):
        return "fails on the wrap"
    return "passes"
