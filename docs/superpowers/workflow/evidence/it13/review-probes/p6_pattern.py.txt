made = []


class Rec(Recorder):
    def __init__(self):
        super().__init__()
        self.levels = []

    def set_brightness(self, level):
        super().set_brightness(level)
        self.levels.append(level)


def fake_make(cfg, on_key=None):
    d = Rec()
    made.append((dict(vars(cfg)), d))
    return d


wp.make_display = fake_make
tmp = Path(tempfile.mkdtemp())


def main(argv):
    made.clear()
    code = wp.main(argv)
    if made:
        cfg, d = made[-1]
        return code, {k: cfg[k] for k in ("backend", "width", "height", "iface", "brightness")}, d
    return code, None, None
