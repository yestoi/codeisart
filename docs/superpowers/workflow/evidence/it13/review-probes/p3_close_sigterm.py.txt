class Lights(FakeLights):
    def __init__(self, on_off=None):
        super().__init__()
        self.offs, self.on_off, self.closed = 0, on_off, False

    def all_off(self):
        self.offs += 1
        super().all_off()
        if self.on_off:
            self.on_off()

    def close(self):
        self.closed = True


class TermDisplay(Recorder):
    """Sends SIGTERM to this process inside the push numbered `at` (a slow push the stop lands in)."""

    def __init__(self, at=None):
        super().__init__()
        self.at, self.calls = at, 0

    def push(self, frame):
        self.calls += 1
        if self.calls == self.at:
            os.kill(os.getpid(), signal.SIGTERM)       # the handler runs here, before the frame is kept
        super().push(frame)


class SteadyPlayer(StrobePlayer):          # the terminal lit once and held: nothing for the governor to hold
    def tick(self, now):
        if self.k == 0:
            self.term.feed(FILL)
        self.k += 1
        return []


def make(tmp, inner, lights):
    sm.make_lights = lambda *a, **k: lights
    sm.make_buttons = lambda *a, **k: None
    sm.AudioCues = lambda *a, **k: FakeAudio()
    loop = ShowLoop(small_cfg(tmp), display=inner, player_factory=SteadyPlayer, notify=lambda s: None)
    t = [0.0]
    loop.clock, loop.sleep = (lambda: t[0]), (lambda s: t.__setitem__(0, t[0] + s))
    return loop


def ctrl_c_at(loop, n):
    """The n-th step raises KeyboardInterrupt, as Ctrl-C (SIGINT) would: the close begins without a SIGTERM."""
    real, k = loop.step, [0]

    def step(now):
        k[0] += 1
        if k[0] == 1:
            loop.presses.put(1)
        real(now)
        if k[0] == n:
            raise KeyboardInterrupt
    loop.step = step


def report(name, loop, inner, lights, outcome):
    pushed = inner.pushed
    lit_last = bool(pushed[-1].any()) if pushed else None
    tail = ["lit" if p.any() else "black" for p in pushed[-3:]]
    print(f"{name}: outcome={outcome}; pushes={len(pushed)} governed={loop.wall.governed if loop.wall else None}; "
          f"last three received={tail}; wall ends lit={lit_last}; display closed={inner.closed}; "
          f"lights offs={lights.offs} modes={set(lights.modes.values()) or '{}'} lights closed={lights.closed}; "
          f"handler after={type(signal.getsignal(signal.SIGTERM)).__name__ if not isinstance(signal.getsignal(signal.SIGTERM), int) else signal.getsignal(signal.SIGTERM)}")


def case(name, n_steps, term_at_push=None, term_in_lights=False):
    with tempfile.TemporaryDirectory() as d:
        tmp = Path(d)
        holder = {}
        fired = [False]

        def lights_term():
            if term_in_lights and not fired[0] and holder.get("closing"):
                fired[0] = True
                os.kill(os.getpid(), signal.SIGTERM)
        lights = Lights(lights_term)
        inner = TermDisplay(term_at_push)
        loop = make(tmp, inner, lights)
        ctrl_c_at(loop, n_steps)
        real_close = loop._close

        def close():
            holder["closing"] = True
            real_close()
        loop._close = close
        try:
            with sigterm_raises():
                outcome = f"run returned {loop.run()}"
        except KeyboardInterrupt:
            outcome = "KeyboardInterrupt out of run"
        report(name, loop, inner, lights, outcome)


# 3a: Ctrl-C at step 10 (pushes 1..10), then the first SIGTERM lands in the close's first black (push 11)
case("A1 SIGTERM in the close's first black", 10, term_at_push=11)
case("A2 SIGTERM in the close's second black", 10, term_at_push=12)
case("A3 SIGTERM in the close's lights.all_off (before the wall)", 10, term_in_lights=True)
case("A0 control: Ctrl-C only", 10)
