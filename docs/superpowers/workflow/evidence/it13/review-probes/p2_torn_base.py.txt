H, W = 64, 128
LIT = np.array((51, 255, 51), np.uint8)


class Screen:
    def __init__(self, fail):
        self.fail, self.count, self.closed = fail, 0, False
        self.screen = np.zeros((H, W, 3), np.uint8)
        self.shown = []

    def push(self, frame):
        self.count += 1
        if self.fail(self.count):
            self.screen[:H // 2] = frame[:H // 2]
            self.shown.append(self.screen.copy())
            raise OSError("torn")
        self.screen = frame.copy()
        self.shown.append(self.screen.copy())

    def set_brightness(self, level):
        pass

    def close(self):
        self.closed = True


def reversal(hz, fps, seconds):
    out = []
    for k in range(int(seconds * fps)):
        f = np.zeros((H, W, 3), np.uint8)
        if round(k / fps * 2 * hz) % 2 == 0:
            f[H // 2:] = LIT
        else:
            f[:H // 2] = LIT
        out.append(f)
    return out


def drive(mod, frames, fps, disp):
    wall = mod.GovernedDisplay(disp, H, W, fps=fps)
    failed, governed = False, []
    try:
        for f in frames:
            try:
                if failed:
                    wall.repush()
                governed.append(wall.push(f).copy())
            except Exception:
                failed = True
                continue
            failed = False
    finally:
        try:
            wall.close()
        except OSError:
            pass
    return governed
