def cells_of(page, frames_h, scale):
    return round((page.height - ss.TITLE_H) / (frames_h * scale + ss.CAP_H + ss.PAD))


print("== pages()")
print(f"{'size':>8} {'n':>3} {'scale':>5} {'cols':>4} {'cap':>5} | pages heights_max over cells_sum lost")
bad = 0
for (h, w) in ((192, 512), (64, 128), (192, 40), (7, 5)):
    for n in (1, 2, 7, 30, 61):
        for scale in (1, 4, 8, 16, 30):
            for cols in (1, 2, 3, 6):
                for cap in (ss.SHOW_MAX_WIDTH, ss.COLS_MAX_WIDTH):
                    frames = [(f"f{k}", np.full((h, w, 3), (k % 250) + 1, np.uint8)) for k in range(n)]
                    got = ss.pages(frames, "plain", scale, "t", 2.2, cols, cap)
                    c, s = fit_width(w, ss.PAD, ss.PAD, cols, scale, cap)
                    # count cells: rows per page * cols (last row may be partial: count from the chunking)
                    rows = max(1, (ss.PAGE_MAX_H - ss.TITLE_H) // (h * s + ss.CAP_H + ss.PAD))
                    per = rows * c
                    total = sum(min(per, n - k * per) for k in range(len(got)))
                    over = [p.height for p in got if p.height > ss.PAGE_MAX_H]
                    heights_ok = all(p.height == ss.TITLE_H + (-(-min(per, n - k * per) // c)) * (h * s + ss.CAP_H
                                     + ss.PAD) for k, p in enumerate(got))
                    if over or total != n or not heights_ok:
                        bad += 1
                        print(f"{w}x{h:<4} {n:>3} {scale:>5} {cols:>4} {cap:>5} | {len(got)} "
                              f"{max(p.height for p in got)} over={over[:3]} cells={total} heights_ok={heights_ok}"
                              f" (fit cols {c} scale {s})")
print("combinations with a page over PAGE_MAX_H or a cell lost:", bad)

# order: the pixel of each page's first cell steps up
