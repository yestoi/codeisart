class Counting:
    def __init__(self):
        self.count, self.closed = 0, False

    def push(self, f):
        self.count += 1

    def set_brightness(self, level):
        pass

    def close(self):
        self.closed = True


import show.wall as sw
real_gov = sw.FlashGovernor


class TinyGuard:                      # refuse at once what would allocate > 1 GB, so the probe does not
    def __init__(self, h, w, *a, **k):   # page the machine; shows what a MemoryError in the governor does
        if h * w > 10_000_000:
            raise MemoryError(f"{h}x{w}")
        self.inner = real_gov(h, w, *a, **k)


cfg, err = cfg_of('backend = "fake"\nfps = 0\nwidth = 100000\nheight = 100000\n')
sw.FlashGovernor = TinyGuard
inner = Counting()
try:
    loop = ShowLoop(cfg, display=inner, notify=lambda s: None, config_error=err)
    loop.start(0.0)
    loop.step(0.05)
    print(f"wall={loop.wall}; display pushes={inner.count}; display closed={inner.closed}; errors={loop.errors}")
    loop._close()
    print(f"after _close: display closed={inner.closed}")
except BaseException as exc:
    print(f"RAISED {type(exc).__name__}: {exc}")
finally:
    sw.FlashGovernor = real_gov
