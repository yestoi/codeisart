    section("B. must-be-zero counts")
    class Failing(Recorder):
        def push(self, frame):
            raise OSError("no carrier")

    class EveryOther(Recorder):
        k = 0

        def push(self, frame):
            self.k += 1
            if self.k % 2 == 0:
                raise OSError("every other")
            super().push(frame)

    for name, disp in (("always failing", Failing()), ("fails every second push", EveryOther())):
        r, _ = run_soak(cfg, seconds=1.0, display=disp, stations=(1,), player_factory=StrobePlayer)
        print(f"{name}: push_failures {r['push_failures']} failures {failures(r)}")

    class TornEveryOther(Recorder):
        """shows the top half of every second frame, then raises: the wall shows a mix."""
        k = 0

        def __init__(self):
            super().__init__()
            self.screen = None
            self.shown = []

        def push(self, frame):
            self.k += 1
            if self.screen is None:
                self.screen = np.zeros_like(frame)
            if self.k % 2 == 0:
                h = frame.shape[0] // 2
                self.screen[:h] = frame[:h]
                self.shown.append(self.screen.copy())
                raise OSError("torn")
            self.screen[:] = frame
            self.shown.append(self.screen.copy())
            super().push(frame)

    torn = TornEveryOther()
    r, _ = run_soak(cfg, seconds=4.0, display=torn, stations=(1,), player_factory=StrobePlayer)
    print(f"torn every other push: push_failures {r['push_failures']} meter squares_max {r['meter']['squares_max']}"
          f" windows squares_max {[w['squares_max'] for w in r['windows']]}; failures {failures(r)}")
    print(f"  the modelled screen: flash_area {flash_area(torn.shown, fps=FPS):.4f} "
          f"square_flashes {square_flashes(torn.shown, fps=FPS)} (budget {BUDGET})")

    # the meter reads the governed frames; with a Recorder, compare to what the display got
    rec = Recorder()
    r, _ = run_soak(cfg, seconds=4.0, display=rec, stations=(1,), player_factory=StrobePlayer)
    print(f"clean strobe soak: meter squares_max {r['meter']['squares_max']} area_max {r['meter']['area_max']};"
          f" display.pushed square_flashes {square_flashes(rec.pushed, fps=FPS)} flash_area "
          f"{flash_area(rec.pushed, fps=FPS)}; meter frames {r['meter']['frames']} pushed {len(rec.pushed)}")
