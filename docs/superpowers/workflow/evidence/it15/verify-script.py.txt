"""Iteration 15's verify runs, from a clean detached checkout of HEAD. Usage: it15-sheets.py sheets|soaks|arcade"""
import os, signal, subprocess, sys, time
S = "/private/tmp/claude-502/-Users-trey-dev-codeisart/d1e01db2-fed8-414f-9a38-325b95bc9c3a/scratchpad"
CO = S + ("/it15-arcade-checkout" if sys.argv[1] == "arcade" else "/it15-verify-checkout")
PY = "/Users/trey/dev/codeisart/.venv/bin/python"
EV = S + "/it15-ev"
os.makedirs(EV, exist_ok=True)
ENV = dict(os.environ, SDL_VIDEODRIVER="dummy", SDL_AUDIODRIVER="dummy")
SHEETS = [
    ("it15-strobe", ["-m", "tools.show_shot", "--session", "strobe", "--seconds", "6", "--every-ms", "150",
                     "--look", "plain", "--scale", "1", "--cols", "4", "--out", EV + "/it15-strobe"]),
    ("it15-presses", ["-m", "tools.show_shot", "--session", "presses", "--every-ms", "1000", "--look", "both",
                      "--out", EV + "/it15-presses"]),
    ("it15-presses-poc", ["-m", "tools.show_shot", "--config", "show.poc.toml", "--session", "presses",
                          "--every-ms", "1000", "--look", "both", "--out", EV + "/it15-presses-poc"]),
]
SOAKS = [
    ("it15-soak", ["-m", "tools.show_soak", "--minutes", "2", "--press-every", "5", "--out", EV + "/it15-soak"]),
    ("it15-soak-poc", ["-m", "tools.show_soak", "--config", "show.poc.toml", "--minutes", "2", "--press-every", "5",
                       "--out", EV + "/it15-soak-poc"]),
]
ARCADE = [
    ("it15-doctor", ["-m", "arcade", "doctor"]),
    ("it15-evidence", ["-m", "tools.arcade_evidence", "--iteration", "15", "--games", "quickdraw,dodge,pong",
                       "--seeds", "20", "--out", EV + "/arcade"]),
    ("it15-shot-quickdraw", ["-m", "tools.arcade_shot", "quickdraw", "--lobby", "small", "--scenario", "canonical",
                             "--raw-vs-pushed", "--out", EV + "/it15-quickdraw"]),
    ("it15-shot-dodge", ["-m", "tools.arcade_shot", "dodge", "--lobby", "small", "--scenario", "canonical",
                         "--raw-vs-pushed", "--out", EV + "/it15-dodge"]),
]


def status():
    return subprocess.run(["git", "status", "--porcelain"], cwd=CO, capture_output=True, text=True).stdout.strip()


def run(runs, logname):
    with open(EV + "/" + logname, "a") as log:
        for name, args in runs:
            t = time.time()
            r = subprocess.run([PY, *args], cwd=CO, env=ENV, capture_output=True, text=True, timeout=580)
            shown = " ".join(a.replace(EV + "/", "") for a in args)
            log.write(f"$ python {shown}\n" + r.stdout)
            if r.stderr:
                log.write("[stderr]\n" + r.stderr[-3000:])
            log.write(f"[exit {r.returncode}, {time.time() - t:.1f} s; git status after: {status() or 'clean'}]\n\n")
            print(name, r.returncode, f"{time.time() - t:.1f}s", "status:", status() or "clean")


def stop():
    with open(EV + "/stop.txt", "w") as log:
        t = time.time()
        p = subprocess.Popen([PY, "-m", "show", "--backend", "fake"], cwd=CO, env=ENV, stdout=subprocess.PIPE,
                             stderr=subprocess.STDOUT, text=True)
        try:
            time.sleep(5)
            p.send_signal(signal.SIGTERM)
            sent = time.time()
            out, _ = p.communicate(timeout=30)
            log.write("$ python -m show --backend fake   (SIGTERM after 5 s)\n" + out[-4000:])
            log.write(f"[exit {p.returncode}, {time.time() - sent:.2f} s after SIGTERM, {time.time() - t:.1f} s in "
                      f"all; git status after: {status() or 'clean'}]\n")
            print("stop", p.returncode, f"{time.time() - sent:.2f}s after SIGTERM")
        finally:
            if p.poll() is None:
                p.kill()
                log.write("[killed by the script: it did not stop in 30 s]\n")
                print("stop: KILLED")


if sys.argv[1] == "sheets":
    run(SHEETS, "show-shot.txt")
    stop()
elif sys.argv[1] == "soaks":
    run(SOAKS, "soak.txt")
else:
    run(ARCADE, "arcade.txt")
