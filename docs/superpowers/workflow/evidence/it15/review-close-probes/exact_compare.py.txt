def named(mod):
    out = {}
    for n in mod.body:
        if isinstance(n, (ast.FunctionDef, ast.ClassDef)):
            out[n.name] = n
        elif isinstance(n, ast.Assign):
            out["=" + ast.unparse(n.targets[0])] = n
        elif isinstance(n, (ast.Import, ast.ImportFrom)):
            out["import " + ast.unparse(n)] = n
        elif isinstance(n, ast.Expr):
            out["doc"] = n
    return out


P, T = named(p), named(t)

    if k not in T:
        print("MISSING in the file:", k)
        diff += 1
    elif ast.dump(n) != ast.dump(T[k]):
        print("DIFFERS:", k)
        diff += 1
    else:
