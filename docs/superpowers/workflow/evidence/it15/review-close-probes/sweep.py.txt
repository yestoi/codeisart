def second(ts):
    if ts == "one":
        return None
    return {"c1": 1, "c2": 2, "new": 3}.get(ts) or int(ts[1:])


class Ticks:
    def __init__(self, inner):
        self.inner, self.slept = inner, []

    def __call__(self):
        return self.inner.tick / fps

    def sleep(self, s):
        self.slept.append(s)
        self.inner.tick += math.ceil(s * fps - 1e-9)


class Track:
    """The measure of square_flashes and flash_area, one slot at a time (fps * K slots a second)."""
    def __init__(self):
        self.px = self.sq = None
        self.max_sq, self.max_area, self.last = 0, 0.0, None
        self.n = fps * K

    def feed(self, frame):
        if frame is self.last and self._repeat_ok:
            s, m = self._s, self._m
        else:
            s = signals(np.asarray(frame), 2.2)
            m = square_means(s, WINDOW)
            self._s, self._m, self.last, self._repeat_ok = s, m, frame, True
        if self.px is None:
            self.px, self.pw = _Transitions(s, THRESHOLD), _Window(self.n, s.shape[1:])
            self.sq, self.sw = _Transitions(m, THRESHOLD), _Window(self.n, m.shape[1:])
            return
        flip, up = self.px.flips(s)
        f = flip.any(axis=0)
        self.max_area = max(self.max_area, float(((self.pw.count - self.pw.ring[self.pw.i] + f) > BUDGET).mean()))
        self.px.advance(s, flip, up)
        self.pw.push(f)
        flip, up = self.sq.flips(m)
        f = flip.any(axis=0)
        self.max_sq = max(self.max_sq, int((self.sw.count - self.sw.ring[self.sw.i] + f).max()))
        self.sq.advance(m, flip, up)
        self.sw.push(f)


def feed_ticks(track, entries, t0, t1, last):
    """Feed ticks t0..t1 (inclusive) of entries [(tick, frame)] into track, K slots a tick; returns the last frame."""
    by = {}
    for tk, f in entries:
        by.setdefault(tk, []).append(f)
    for tk in range(t0, t1 + 1):
        fr = by.get(tk) or [last]
        if len(fr) > K:
            raise RuntimeError(f"{len(fr)} calls in tick {tk}")
        for f in fr + [fr[-1]] * (K - len(fr)):
            track.feed(f)
        last = fr[-1]
    return last
