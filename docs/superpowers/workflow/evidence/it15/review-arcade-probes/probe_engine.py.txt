def run(dets_at, seconds=4.0):
    """dets_at(capture index) -> detections; 10 captures a second, 30 ticks; returns per-tick (value, body)."""
    tr, depth = BodyTracker(), Depth()
    out, body, cam = [], None, 0.0
    for i in range(round(seconds * 30)):
        t = i / 30
        if i % 3 == 0:
            cam = t
            bodies = tr.update(dets_at(i // 3), cam)
            body = bodies[0] if bodies else None
        out.append((depth.update(body, t, cam), body))
    return out

def sideways(c):
    box, kps = spike_det(hips=c < 6)
    if c == 0:
        kps = list(kps)
        mx = (kps[LEFT_SHOULDER].x + kps[RIGHT_SHOULDER].x) / 2
        for i in (LEFT_SHOULDER, RIGHT_SHOULDER):
            kps[i] = Keypoint(mx + (kps[i].x - mx) * 0.15, kps[i].y, kps[i].conf)
        kps = tuple(kps)
    # a wrist 0.2 above the shoulders
    kps = list(kps)
    sy = kps[LEFT_SHOULDER].y
    kps[RIGHT_WRIST] = Keypoint(kps[RIGHT_SHOULDER].x + 0.05, sy - 0.2, 1.0)
    return [(box, tuple(kps))]

def dup(c, shift=0.015):
    a = spike_det()
    box, kps = spike_det(cx=0.5 + shift)
    hc = (0.9, 0.8) if c % 2 else (0.8, 0.9)
    k1 = list(a[1]); k2 = list(kps)
    for i in (LEFT_HIP, RIGHT_HIP):
        k1[i] = Keypoint(k1[i].x, k1[i].y, hc[0]); k2[i] = Keypoint(k2[i].x, k2[i].y, hc[1])
    return merge_duplicates([(a[0], tuple(k1)), (box, tuple(k2))])


tr = BodyTracker()
xs, zs, ids, n = [], [], set(), []
for c in range(40):
    bodies = tr.update(dup(c), c / 10)
    n.append(len(bodies))
    ids |= {b.id for b in bodies}
    xs.append(bodies[0].shoulder_mid.x)
    zs.append(bodies[0].zone_x)
steps = [abs(b - a) for a, b in zip(xs[10:], xs[11:])]
print(f"   bodies per capture {set(n)}, ids {ids}, shoulder x step per capture after 1 s: max {max(steps):.4f} "
      f"mean {sum(steps) / len(steps):.4f}; zone_x range {min(zs[10:]):.4f} .. {max(zs[10:]):.4f}")
