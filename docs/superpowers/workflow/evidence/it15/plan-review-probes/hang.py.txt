# A real process: the planned close, real time.sleep, the show's Sigterm handler already seen once (the stop).
from show.main import Sigterm  # noqa: E402
from tests.test_wall_hold import Clocked, H, W, strobe  # noqa: E402

wall_p.MODE = "plan"
handler = Sigterm()
handler.seen = 1                                   # the first SIGTERM started the close
signal.signal(signal.SIGTERM, handler)
now = [0.0]
inner = Clocked({3}, lambda: now[0])
wall = wall_p.GovernedDisplay(inner, H, W, fps=16, from_dark=True, clock=lambda: now[0])   # default: time.sleep
fr = strobe(1, 16, 3)
wall.push(fr[0]), wall.push(fr[1])
now[0] = 2 / 16
try:
    wall.push(fr[2])
except OSError:
    pass
now[0] = 3 / 16
threading.Timer(0.3, lambda: os.kill(os.getpid(), signal.SIGTERM)).start()
t0 = time.monotonic()
wall.close()
print(f"second SIGTERM in the wait: close took {time.monotonic() - t0:.2f} s (planned 15/16 + 1 = 1.94), "
      f"seen {handler.seen}, calls {inner.calls}, closed {inner.closed}, last black {not inner.sent[-1][1].any()}")

signal.signal(signal.SIGINT, signal.default_int_handler)
inner = Clocked({3}, lambda: now[0])
wall = wall_p.GovernedDisplay(inner, H, W, fps=16, from_dark=True, clock=lambda: now[0])
now[0] = 0.0
wall.push(fr[0]), wall.push(fr[1])
now[0] = 2 / 16
try:
    wall.push(fr[2])
except OSError:
    pass
now[0] = 3 / 16
threading.Timer(0.3, lambda: os.kill(os.getpid(), signal.SIGINT)).start()
t0 = time.monotonic()
try:
    wall.close()
    r = "no raise"
except KeyboardInterrupt:
    r = "KeyboardInterrupt"
print(f"Ctrl-C in the wait: {r} after {time.monotonic() - t0:.2f} s, calls {inner.calls} (3: nothing after the "
      f"tear), closed {inner.closed}, the wall left as the tear left it")
