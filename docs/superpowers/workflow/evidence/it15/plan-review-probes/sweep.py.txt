def after(seq, m, start):
    """square_flashes and flash_area, only the windows that end at slot `start` or later (the tracker warmed up
    on the second before them: a tracker started cold on a slice can count a swing the whole run does not)."""
    sq, area = 0, 0.0
    for i, (flip, up, c) in enumerate(_counted(seq, 2.2, m, THRESHOLD, squares=32)):
        if i + 1 >= start:
            sq = max(sq, int((c.count - c.ring[c.i] + flip.any(axis=0)).max()))
    for i, (flip, up, c) in enumerate(_counted(seq, 2.2, m, THRESHOLD)):
        if i + 1 >= start:
            area = max(area, float(((c.count - c.ring[c.i] + flip.any(axis=0)) > BUDGET).mean()))
    return sq, area

class Ticks:
    def __init__(self, inner):
        self.inner, self.slept = inner, []

    def __call__(self):
        return self.inner.tick / fps

    def sleep(self, s):
        self.slept.append(s)
        self.inner.tick += math.ceil(s * fps - 1e-9)


def measure(w, t):
    d = w.display
    before = len(d.shown)
    raised = None
    try:
        w.close()
    except BaseException as e:  # noqa: B036
        raised = type(e).__name__
    new = d.shown[before:]
    lit = bool(d.screen.any() or d.rows.any())
    if not new:
        return None, (0, 0.0, lit, d.closed, sum(t.slept), raised)
    s0 = max(0, new[0][0] - 2 * fps)
    prior = DARK
    for tk, f in d.shown:
        if tk < s0:
            prior = f
    ents = [(0, prior)] + [(tk - s0 + 1, f) for tk, f in d.shown if tk >= s0]
    h = hashlib.md5(repr([tk for tk, _ in ents]).encode())
    for _, f in ents:
        h.update(f.tobytes())
    key = h.hexdigest()
    if key in CACHE:
        sq, area = CACHE[key]
    else:
        seq, m = in_time(ents, fps)
        seq = seq[m // fps:]
        sq, area = after(seq, m, (new[0][0] - s0 + 1) * (m // fps))
        CACHE[key] = (sq, area)
    return key, (sq, area, lit, d.closed, sum(t.slept), raised)
