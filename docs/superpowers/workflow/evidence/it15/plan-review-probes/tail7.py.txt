import numpy as np  # noqa: E402

from show.wall import GovernedDisplay  # noqa: E402
from tests.test_wall_hold import TornDisplay, in_time  # noqa: E402

fps, pic = 20, pics.PICS["rev rows b32 ph1 5Hz"]


def run(tears, kc, card=False):
    inner = TornDisplay(lambda c: c in tears, split=33, card=card)
    wall = GovernedDisplay(inner, 64, 128, fps=fps, from_dark=True, clock=lambda: inner.tick / fps)
    for k in range(kc):
        inner.tick = k
        try:
            wall.push(pic(k, fps))
        except OSError:
            pass
    inner.tick = kc
    holding = wall.holding
    wall.close()
    seq, n = in_time(inner.shown, fps)
    return seq, n, holding, inner


for tears, label in (({12}, "tear at call 12"), (set(), "no tear")):
    worst = []
    for kc in range(20, 110):
        seq, n, holding, inner = run(tears, kc)
        worst.append((square_flashes(seq, fps=n), kc, holding))
    over = [(kc, h) for s, kc, h in worst if s > 6]
    print(f"{label}: close at tick 20..109 (today's close): max squares {max(w[0] for w in worst)}; over 6 at ticks "
          f"{over}")
seq, n, holding, inner = run({12}, 93)
k = n // fps
flips = [f.any(axis=0) for f, up, w in _counted(seq, 2.2, n, 0.1, squares=32)]
tot = np.array(flips)                       # (slots-1, squares)
best = None
for end in range(len(tot)):
    c = tot[max(0, end - n + 1):end + 1].sum(axis=0)
    if best is None or c.max() > best[0]:
        best = (int(c.max()), end, np.unravel_index(c.argmax(), c.shape))
print("close at tick 93, tear at call 12: slots per tick", k, "worst", best[0], "window ending at slot", best[1],
      "(tick", best[1] // k, ") square", best[2])
sq = best[2]

print("that square's transitions in the window, by tick:", ticks)
