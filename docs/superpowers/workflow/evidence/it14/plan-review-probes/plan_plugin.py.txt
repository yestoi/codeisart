def pytest_configure(config):
    import show.main as main
    import show.wall as wall
    import wall14
    wall14.HOLD_S = wall.HOLD_S = 1.0
    wall14.FlashGovernor = lambda *a, **k: wall.FlashGovernor(*a, **k)   # the tests monkeypatch show.wall's
    wall.GovernedDisplay = wall14.GovernedDisplay
    main.GovernedDisplay = wall14.GovernedDisplay
    L = main.ShowLoop
    start0, step0, open0 = L.start, L.step, L._open_wall
    init0 = L.__init__

    def __init__(self, *a, **kw):
        init0(self, *a, **kw)
        self._now = 0.0

    def start(self, now):
        self._now = now
        return start0(self, now)

    def step(self, now):
        self._now = now
        return step0(self, now)

    def _open_wall(self):
        main.GovernedDisplay = lambda *a, **kw: wall14.GovernedDisplay(*a, from_dark=True,
                                                                        clock=lambda: self._now, **kw)
        try:
            return open0(self)
        finally:
            main.GovernedDisplay = wall14.GovernedDisplay

    def _push(self, frame, now):
        if self.wall is None:
            self._failing(now)
            return
        if frame is None:
            return
        try:
            sent = self.wall.push(frame)
        except Exception:
            self._failures += 1
            if self._failures == 1 or self._failures % main.FAILURE_LOG_EVERY == 0:
                main.log.exception("the push failed (%d in a row); the show goes on", self._failures)
            self._failing(now)
            return
        if sent is None:
            self._failing(now)
            return
        if self._failures:
            main.log.info("the push works again after %d failures", self._failures)
        self._failures, self._fail_since = 0, None
        if self._dark:
            self._dark = False
            if self.show is not None:
                try:
                    self.show.relight()
                except Exception:
                    main.log.exception("relighting failed")

    L.__init__, L.start, L.step, L._open_wall, L._push = __init__, start, step, _open_wall, _push
