"""Q1: the planned quiet hold (the writer's scratch wall, mode quiet) against pictures and tears the writer did not
try: other boundaries, a left/right reversal, 32x32 checks at two offsets, a partial strobe, a slow scroll, a
scrolling '#' band; one tear at call n (1..22) at splits 8/20/33/40/56; two tears (n, n+j), j = 1..4 (a tear on a
counted send and on the first new frame); fps 20 and 30; card and probe models. The worst square count and area."""
import itertools
import sys as _s
import time
import numpy as np
from rlib import LIT, Torn, drive, measure

H, W = 64, 128
FPSES = [int(a) for a in _s.argv[1].split(",")] if len(_s.argv) > 1 else [20, 30]


def per(fps, hz, k):
    return round(k / fps * 2 * hz) % 2


def rev_rows(b, hz, phase):
    def f(k, fps):
        x = np.zeros((H, W, 3), np.uint8)
        if per(fps, hz, k) ^ phase:
            x[:b] = LIT
        else:
            x[b:] = LIT
        return x
    return f


def rev_cols(hz, phase):
    def f(k, fps):
        x = np.zeros((H, W, 3), np.uint8)
        if per(fps, hz, k) ^ phase:
            x[:, :W // 2] = LIT
        else:
            x[:, W // 2:] = LIT
        return x
    return f


def checks(off, hz, phase):
    yy, xx = np.mgrid[0:H, 0:W]
    m = (((yy + off) // 32 + (xx + off) // 32) % 2).astype(bool)
    def f(k, fps):
        x = np.zeros((H, W, 3), np.uint8)
        x[m if per(fps, hz, k) ^ phase else ~m] = LIT
        return x
    return f


def block_strobe(hz, phase):
    def f(k, fps):
        x = np.zeros((H, W, 3), np.uint8)
        if per(fps, hz, k) ^ phase:
            x[8:56, 40:88] = LIT
        return x
    return f


def scroll(step):
    def f(k, fps):
        x = np.zeros((H, W, 3), np.uint8)
        y = (k * step) % (H + 8) - 8
        x[max(0, y):max(0, y + 8)] = LIT
        return x
    return f


def hashes(k, fps):
    x = np.zeros((H, W, 3), np.uint8)
    yy, xx = np.mgrid[0:8, 0:W]
    m = ((xx + k) % 6 < 3) & (yy % 4 != 3)       # a band of '#'-like marks scrolling 1 px a tick
    x[28:36][m] = LIT
    return x


PICS = {}
for b in (16, 32, 48):
    for ph in (0, 1):
        PICS[f"rev rows b{b} ph{ph} 10Hz"] = rev_rows(b, 10, ph)
PICS["rev rows b32 ph1 5Hz"] = rev_rows(32, 5, 1)
PICS["rev rows b24 ph1 5Hz"] = rev_rows(24, 5, 1)
for ph in (0, 1):
    PICS[f"rev cols ph{ph} 10Hz"] = rev_cols(10, ph)
for off in (0, 16):
    PICS[f"checks off{off} 10Hz"] = checks(off, 10, 0)
    PICS[f"checks off{off} 5Hz ph1"] = checks(off, 5, 1)
PICS["block strobe 10Hz"] = block_strobe(10, 0)
PICS["block strobe 5Hz ph1"] = block_strobe(5, 1)
PICS["scroll 1 row"] = scroll(1)
PICS["scroll 3 rows"] = scroll(3)
PICS["hash band"] = hashes

TEARS = [("one", n, None, s) for n in range(1, 23, 1) for s in (8, 20, 33, 56)]
TEARS += [("two", n, j, s) for n in (3, 6, 7, 9) for j in (1, 2, 3, 4) for s in (24, 40)]

t0 = time.time()
print("fps | model | picture | worst sq (area) at (kind, n, j, split) | cases")
for fps in FPSES:
    for model in ("card", "probe"):
        for name, pic in PICS.items():
            worst, cases = (-1, 0.0, None), 0
            for kind, n, j, split in TEARS:
                if kind == "one":
                    tears = (lambda c, n=n, s=split: s if c == n else 0)
                else:
                    tears = (lambda c, n=n, j=j, s=split: s if c in (n, n + j) else 0)
                ticks = n + (j or 0) + 4 * fps + 5
                d = Torn(H, W, tears, split, model, fps=fps)
                drive([pic(k, fps) for k in range(ticks)], d, fps)
                a, sq = measure(d, ticks, fps)
                cases += 1
                if (sq, a) > (worst[0], worst[1]):
                    worst = (sq, a, (kind, n, j, split))
            print(f"{fps} | {model} | {name} | {worst[0]} ({worst[1]:.3f}) at {worst[2]} | {cases}", flush=True)
print(f"seconds {time.time() - t0:.0f}")
