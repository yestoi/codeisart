H, Wd = 64, 128
LIT = np.array((51, 255, 51), np.uint8)


class Rec:
    def __init__(self, fail, clock):
        self.fail, self.clock, self.calls, self.sent = set(fail), clock, 0, []

    def push(self, f):
        self.calls += 1
        if self.calls in self.fail:
            raise OSError("x")
        self.sent.append((self.clock(), f.copy()))

    def set_brightness(self, l):
        pass

    def close(self):
        pass


def frames(n):
    return [np.broadcast_to(LIT if k % 2 == 0 else np.zeros(3, np.uint8), (H, Wd, 3)).copy() for k in range(n)]
