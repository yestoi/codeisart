def status():
    return subprocess.run(["git", "status", "--porcelain"], cwd=CO, capture_output=True, text=True).stdout.strip()


def run(runs, logname):
    with open(EV + "/" + logname, "a") as log:
        for name, args in runs:
            t = time.time()
            r = subprocess.run([PY, *args], cwd=CO, env=ENV, capture_output=True, text=True, timeout=500)
            shown = " ".join(a.replace(EV + "/", "") for a in args)
            log.write(f"$ python {shown}\n" + r.stdout)
            if r.stderr:
                log.write("[stderr]\n" + r.stderr[-3000:])
            log.write(f"[exit {r.returncode}, {time.time() - t:.1f} s; git status after: {status() or 'clean'}]\n\n")
            print(name, r.returncode, f"{time.time() - t:.1f}s", "status:", status() or "clean")


def stop():
    with open(EV + "/stop.txt", "w") as log:
        t = time.time()
        p = subprocess.Popen([PY, "-m", "show", "--backend", "fake"], cwd=CO, env=ENV, stdout=subprocess.PIPE,
                             stderr=subprocess.STDOUT, text=True)
        try:
            time.sleep(5)
            p.send_signal(signal.SIGTERM)
            sent = time.time()
            out, _ = p.communicate(timeout=30)
            log.write("$ python -m show --backend fake   (SIGTERM after 5 s)\n" + out[-4000:])
            log.write(f"[exit {p.returncode}, {time.time() - sent:.2f} s after SIGTERM, {time.time() - t:.1f} s in "
                      f"all; git status after: {status() or 'clean'}]\n")
            print("stop", p.returncode, f"{time.time() - sent:.2f}s after SIGTERM")
        finally:
            if p.poll() is None:
                p.kill()
                log.write("[killed by the script: it did not stop in 30 s]\n")
                print("stop: KILLED")


if sys.argv[1] == "sheets":

    stop()
