"""Read-only: the proposed fix for the loop strobe test (the StrobePlayer hides the cursor with ESC[?25l),
and the alternative assert (concurrent_area under SMALL_AREA) on the unfixed input."""
from pathlib import Path
from arcade.flash import SMALL_AREA, BUDGET, FlashGovernor, concurrent_area, flash_area, square_flashes
from show.font import Font
from show.renderer import Renderer
from show.terminal import Terminal

ROOT = Path("/Users/trey/dev/codeisart")
H, W = 192, 512
FILL = b"\x1b[H\x1b[7m" + b" " * (80 * 23 - 1) + b"\x1b[0m"
CLEAR = b"\x1b[H\x1b[2J"
font = Font.load(ROOT / "fonts" / "5x7.bin")
for name, on, off in (("plan's feeds", FILL, CLEAR), ("ESC[?25l first", b"\x1b[?25l" + FILL, b"\x1b[?25l" + CLEAR)):
    r = Renderer(font, W, H, 80, 24, (51, 255, 51), False, "text")
    term = Terminal(80, 23); term.reset(23)
    rendered = []
    for k in range(1, 101):
        now = k / 20
        term.feed(on if (k - 1) % 2 == 0 else off)
        strip = "NOW: a by Test Author, 2026, Not A.I." + (" | PLAYING" if now < 2.0 else "")
        rendered.append(r.render(term.screen, int(now * 2) % 2 == 0, strip).copy())
    g = FlashGovernor(H, W, 2.2, fps=20)
    pushed = [g.apply(f).copy() for f in rendered]
    print(f"{name:15s}: raw area {flash_area(rendered, fps=20):.3f} pushed area {flash_area(pushed, fps=20):.5f} "
          f"concurrent {concurrent_area(pushed, fps=20):.4f} (SMALL_AREA {SMALL_AREA}) "
          f"square_flashes {square_flashes(pushed, fps=20)} (BUDGET {BUDGET}) held {g.held_ticks}")
